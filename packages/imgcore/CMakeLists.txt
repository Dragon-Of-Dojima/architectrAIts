cmake_minimum_required(VERSION 3.16)
project(imgcore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(imgcore STATIC dhash.cpp)
target_include_directories(imgcore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
enable_testing()
add_executable(testdhash testdhash.cpp)
target_link_libraries(testdhash PRIVATE imgcore)
add_test(NAME dhash_basic COMMAND testdhash)
